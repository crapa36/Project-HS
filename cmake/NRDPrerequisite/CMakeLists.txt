cmake_minimum_required(VERSION 3.22...4.4)
project(HsNrdPrerequisite LANGUAGES C CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR CMAKE_SYSTEM_PROCESSOR MATCHES "ARM|arm")
    message(FATAL_ERROR "The pinned NRD prerequisite is Windows x64 only.")
endif()
foreach(path IN ITEMS HS_NRD_SOURCE HS_SHADERMAKE_SOURCE HS_MATHLIB_SOURCE HS_NRD_DXC)
    if(NOT EXISTS "${${path}}")
        message(FATAL_ERROR "Missing offline prerequisite: ${path}")
    endif()
endforeach()
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "" FORCE)
set(SHADERMAKE_FIND_COMPILERS OFF CACHE BOOL "" FORCE)
set(SHADERMAKE_FIND_DXC OFF CACHE BOOL "" FORCE)
set(SHADERMAKE_FIND_FXC OFF CACHE BOOL "" FORCE)
set(SHADERMAKE_FIND_DXC_VK OFF CACHE BOOL "" FORCE)
set(SHADERMAKE_FIND_SLANG OFF CACHE BOOL "" FORCE)
set(SHADERMAKE_TOOL OFF CACHE BOOL "" FORCE)
set(SHADERMAKE_DXC_PATH "${HS_NRD_DXC}" CACHE FILEPATH "" FORCE)
set(NRD_NRI OFF CACHE BOOL "" FORCE)
set(NRD_STATIC_LIBRARY OFF CACHE BOOL "" FORCE)
set(NRD_NORMAL_ENCODING 2 CACHE STRING "" FORCE)
set(NRD_ROUGHNESS_ENCODING 1 CACHE STRING "" FORCE)
set(NRD_EMBEDS_DXIL_SHADERS ON CACHE BOOL "" FORCE)
set(NRD_EMBEDS_DXBC_SHADERS OFF CACHE BOOL "" FORCE)
set(NRD_EMBEDS_SPIRV_SHADERS OFF CACHE BOOL "" FORCE)
set(NRD_SUPPORTS_VIEWPORT_OFFSET OFF CACHE BOOL "" FORCE)
set(NRD_SUPPORTS_CHECKERBOARD ON CACHE BOOL "" FORCE)
set(NRD_SUPPORTS_HISTORY_CONFIDENCE ON CACHE BOOL "" FORCE)
set(NRD_SUPPORTS_DISOCCLUSION_THRESHOLD_MIX ON CACHE BOOL "" FORCE)
set(NRD_SUPPORTS_ANTIFIREFLY ON CACHE BOOL "" FORCE)
set(REBLUR_PERFORMANCE_MODE OFF CACHE BOOL "" FORCE)
set(NRD_SHADERS_PATH "${CMAKE_CURRENT_BINARY_DIR}/shaders" CACHE STRING "" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
add_subdirectory("${HS_SHADERMAKE_SOURCE}" shadermake)
add_subdirectory("${HS_MATHLIB_SOURCE}" mathlib)
add_subdirectory("${HS_NRD_SOURCE}" nrd)
